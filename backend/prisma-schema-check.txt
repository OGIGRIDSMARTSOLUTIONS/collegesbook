model User {
  id                     String             @id @default(cuid())
  email                  String?            @unique
  phone                  String?            @unique
  passwordHash           String
  role                   Role               @default(STUDENT)
  accountStatus          AccountStatus      @default(PENDING)
  emailVerifiedAt        DateTime?
  lastLoginAt            DateTime?
  refreshTokenHash       String?
  createdAt              DateTime           @default(now())
  updatedAt              DateTime           @updatedAt
  username               String?            @unique
  passwordResetTokenHash String?            @unique
  passwordResetExpiresAt DateTime?
  auditLogs              AuditLog[]
  institutionEvents      InstitutionEvent[] @relation("InstitutionEventCreator")
  institutionStaff       InstitutionStaff?
  student                Student?

  @@index([role])
  @@map("users")
}

model Student {

  bio                 String?
  gender              String?

  phoneNumber         String?

model Post {
  id             String         @id @default(cuid())
  authorId       String
  institutionId  String
  body           String?
  images         Json?
  linkUrl        String?
  visibility     PostVisibility @default(INSTITUTION)
  createdAt      DateTime       @default(now())
  updatedAt      DateTime       @updatedAt
  deletedAt      DateTime?
  groupId        String?
  repostOfId     String?
  comments       Comment[]
  reposts        PostRepost[]
  author         Student        @relation(fields: [authorId], references: [id], onDelete: Cascade)
  group          Group?         @relation(fields: [groupId], references: [id], onDelete: Cascade)
  repostOf       Post?          @relation("PostRepostOrigin", fields: [repostOfId], references: [id])
  repostedCopies Post[]         @relation("PostRepostOrigin")
  reactions      Reaction[]
  reports        Report[]
  savedBy        SavedPost[]

  @@index([authorId])
  @@index([institutionId, createdAt])
  @@index([groupId])
  @@index([authorId, createdAt])
  @@index([repostOfId])
  @@map("posts")
}

  @@index([studentId, createdAt])
